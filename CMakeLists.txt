cmake_minimum_required(VERSION 3.20)
project(MeshCator VERSION 0.3.25 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(meshcator_core src/core/App.cpp src/core/I18n.cpp src/core/Configuration.cpp src/core/SettingsFlow.cpp src/core/MapFlow.cpp src/core/ChannelFlow.cpp src/core/Notifications.cpp src/core/History.cpp src/core/Management.cpp)
target_include_directories(meshcator_core PUBLIC src)
target_compile_options(meshcator_core PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE meshcator_core)
add_test(NAME core COMMAND core_tests)
add_executable(settings_tests tests/settings_tests.cpp src/mesh/SimBackend.cpp)
target_link_libraries(settings_tests PRIVATE meshcator_core)
add_test(NAME settings COMMAND settings_tests)
add_executable(configuration_tests tests/configuration_tests.cpp)
target_link_libraries(configuration_tests PRIVATE meshcator_core)
add_test(NAME configuration COMMAND configuration_tests)
add_executable(notification_tests tests/notification_tests.cpp)
target_link_libraries(notification_tests PRIVATE meshcator_core)
add_test(NAME notifications COMMAND notification_tests)
find_package(ZLIB REQUIRED)
add_executable(map_tiles_tests tests/map_tiles_tests.cpp)
target_include_directories(map_tiles_tests PRIVATE src)
# Avoid adding the macOS SDK C headers ahead of libc++ headers.
if(NOT APPLE)
 target_include_directories(map_tiles_tests PRIVATE ${ZLIB_INCLUDE_DIRS})
endif()
target_link_libraries(map_tiles_tests PRIVATE ${ZLIB_LIBRARIES})
add_test(NAME map_tiles COMMAND map_tiles_tests)
add_executable(history_tests tests/history_tests.cpp src/mesh/SimBackend.cpp)

add_test(NAME advert_push COMMAND advert_push_tests)
add_executable(usb_routing_tests tests/usb_routing_tests.cpp)
target_include_directories(usb_routing_tests PRIVATE tests/stubs vendor/meshcore-pager/src)
target_compile_definitions(usb_routing_tests PRIVATE MESHCATOR_UI=1)
add_test(NAME usb_isolation COMMAND usb_routing_tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME usb_client COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_usb_client.py)
option(MESHCATOR_SIMULATOR "Build LVGL/SDL desktop simulator" ON)
if(MESHCATOR_SIMULATOR)
 include(FetchContent)
 set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/config/lv_conf.h" CACHE STRING "" FORCE)
 set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
 set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
 set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
 FetchContent_Declare(lvgl GIT_REPOSITORY https://github.com/lvgl/lvgl.git GIT_TAG v9.2.2 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(lvgl)
 find_package(SDL2 REQUIRED CONFIG)
 find_package(ZLIB REQUIRED)
 file(GLOB MESHCATOR_FONTS CONFIGURE_DEPENDS src/ui/fonts/*.c)
 add_executable(meshcator desktop/main.cpp src/ui/View.cpp src/ui/MapView.cpp src/mesh/SimBackend.cpp ${MESHCATOR_FONTS})
 target_link_libraries(meshcator PRIVATE meshcator_core lvgl::lvgl SDL2::SDL2 ${ZLIB_LIBRARIES})
 if(NOT APPLE)
  target_include_directories(meshcator PRIVATE ${ZLIB_INCLUDE_DIRS})
 endif()
 target_compile_options(meshcator PRIVATE -Wall -Wextra -Wpedantic)
 add_test(NAME emoji_render_font COMMAND meshcator --smoke --emoji-test --settings ${CMAKE_CURRENT_BINARY_DIR}/emoji-font-settings.txt)
 set_tests_properties(emoji_render_font PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
 add_test(NAME full_name_layout COMMAND meshcator --smoke --name-layout-test --settings ${CMAKE_CURRENT_BINARY_DIR}/name-layout-settings.txt)
 set_tests_properties(full_name_layout PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
 add_test(NAME map_render_cache COMMAND meshcator --smoke --map-perf --settings ${CMAKE_CURRENT_BINARY_DIR}/map-perf-settings.txt)
 set_tests_properties(map_render_cache PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
 foreach(screen RANGE 0 8)
  add_test(NAME render_${screen} COMMAND meshcator --smoke --screen ${screen}
           --settings ${CMAKE_CURRENT_BINARY_DIR}/smoke-settings.txt)
  set_tests_properties(render_${screen} PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
 endforeach()
endif()
